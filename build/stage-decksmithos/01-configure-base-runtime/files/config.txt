# DecksmithOS hardware profile
# No initramfs: the kernel mounts the root (NVMe and ext4 are built in) and
# decksmithos-overlay-init (init= in cmdline.txt) sets up the read-only root.
# Do not add auto_initramfs=1: the initramfs would run first and add ~1.3 s.
disable_overscan=1
usb_max_current_enable=1
# PCIe Gen 3 is not officially certified on Pi 5 / CM5 (Gen 2 is).
# Keep only if validated on the exact carrier + SSD: link at 8GT/s,
# no AER/NVMe errors in dmesg under sustained load. Otherwise remove
# this line to fall back to Gen 2. See AGENTS.md section 12.
dtparam=pciex1_gen=3

# RTC battery: charging stays off (firmware default). The development board
# has a CR2032, which must never be charged. Add dtparam=rtc_bbat_vchg=3000000
# only on a board fitted with a rechargeable cell (official ML-2020 Li-Mn).

# Fan connector (J14, 4 pins: 5 V, GND, PWM, tach) of the CM5 IO board: the
# kernel's pwm-fan driver runs a 5 V PWM fan by the SoC temperature (same
# trip points as the Pi 5). The enclosure has a 40 mm exhaust fan.
dtparam=cooling_fan=on

# disable onboard wifi
dtoverlay=disable-wifi

# disable onboard bluetooth
dtoverlay=disable-bt

# disable onboard audio
dtparam=audio=off

# noaudio: no HDMI audio devices; audio goes only through the controller.
dtoverlay=vc4-kms-v3d,noaudio
